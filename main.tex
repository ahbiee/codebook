\subsection{模板不能直接套時 (題目變形)}
\begin{enumerate}
    \item \textbf{多了一個條件} $\rightarrow$ 把它塞進「狀態」：最短路變成 (點, 條件) 的分層圖；DP 多開一維。狀態數 $\times$ 轉移數 $\le 10^8$ 就能做。
    \item \textbf{終點固定、起點很多} $\rightarrow$ 反向建圖從終點跑一次；\textbf{起點很多} $\rightarrow$ 超級源點或多源 BFS。
    \item \textbf{直接求很難、但「猜答案 $x$」後很好檢查} $\rightarrow$ 二分搜答案。
    \item \textbf{換個問法}：總數 $-$ 不合法數；最小割 = 最大流；最小點覆蓋 = 最大匹配；最大獨立集 = $N -$ 匹配。
    \item \textbf{換枚舉的東西}：枚舉答案、枚舉中間點、枚舉比較小的那一維；「所有子陣列的總和」$\rightarrow$ 算每個元素被算幾次 (貢獻法)。
    \item \textbf{離線處理}：把詢問排序；刪除很難 $\rightarrow$ 倒過來變加入 (DSU)。
    \item \textbf{化簡}：先排序 (很多東西會變單調)；$|x_1-x_2|+|y_1-y_2|$ 轉成 $(x+y, x-y)$ 取 max；值域大但個數少 $\rightarrow$ 座標壓縮。
    \item \textbf{完全沒想法}：寫暴力把小範圍答案印出來找規律 (打表)，再用對拍驗證。
\end{enumerate}

\subsection{常用公式}
\begin{itemize}
    \item Catalan: $C_n=\frac{1}{n+1}\binom{2n}{n}$，1, 1, 2, 5, 14, 42, 132。合法括號、二元樹個數、不越過對角線的路徑。
    \item 第二類 Stirling (n 個相異球放進 k 個相同非空盒): $S(n,k)=kS(n-1,k)+S(n-1,k-1)$。
    \item 錯排: $D_n=(n-1)(D_{n-1}+D_{n-2})$，$D_1=0, D_2=1$。
    \item 插板法: $x_1+\dots+x_k=n$ 的非負整數解 $\binom{n+k-1}{k-1}$；正整數解 $\binom{n-1}{k-1}$。
    \item 格子路徑 $(0,0)\to(a,b)$ 只往右/上: $\binom{a+b}{a}$。
    \item $\sum i^2=\frac{n(n+1)(2n+1)}{6}$，$\sum i^3=\left(\frac{n(n+1)}{2}\right)^2$，等比 $\sum_{i=0}^{n-1} r^i=\frac{r^n-1}{r-1}$。
    \item 費馬: $a^{p-1}\equiv1$；尤拉: $a^{\varphi(m)}\equiv1$ ($\gcd(a,m)=1$)；Wilson: $(p-1)!\equiv-1 \pmod p$。
    \item $\gcd(a,b)=\gcd(b,a\bmod b)$，$\text{lcm}(a,b)=a/\gcd(a,b)\cdot b$ (先除再乘防溢位)。
    \item 平面圖 Euler: $V-E+F=2$ (連通，F 含外部面)。Pick: $A=I+B/2-1$。
    % [未學] \item Möbius: $\sum_{d\mid n}\mu(d)=[n=1]$；$1\le i,j\le n$ 中 $\gcd(i,j)=1$ 的對數 $=\sum_{d}\mu(d)\lfloor n/d\rfloor^2$ (配數論分塊)。
    % [未學] \item Burnside (旋轉後相同算同一種): 方法數 $=\frac{1}{n}\sum_{i=0}^{n-1} k^{\gcd(i,n)}$ (長度 $n$ 的項鍊、$k$ 種顏色)。
\end{itemize}

\section{幾何}

\subsection*{A. 點與直線}
點 $(x_0,y_0)$ 到直線 $ax+by+c=0$ 的距離 $\frac{|ax_0+by_0+c|}{\sqrt{a^2+b^2}}$；
兩平行線 $ax+by+c_1=0$、$ax+by+c_2=0$ 距離 $\frac{|c_1-c_2|}{\sqrt{a^2+b^2}}$。

\subsection*{B. 三角形}
頂點 $A(x_1,y_1), B(x_2,y_2), C(x_3,y_3)$，對邊長 $a,b,c$，面積 $\Delta$。
\[
\Delta = \frac{1}{2} \left|\begin{vmatrix} x_1 & y_1 & 1 \\ x_2 & y_2 & 1 \\ x_3 & y_3 & 1 \end{vmatrix}\right|,\quad
\Delta=\sqrt{s(s-a)(s-b)(s-c)},\ s=\tfrac{a+b+c}{2}
\]
重心 $G=\frac{A+B+C}{3}$；內心 $I=\frac{aA+bB+cC}{a+b+c}$；外心 $O$ 見 Circle.cpp；
垂心 $H = 3G - 2O$ (重心、外心、垂心共線)。
\[
O_x = \frac{1}{4\Delta} \begin{vmatrix} x_1^2 + y_1^2 & y_1 & 1 \\ x_2^2 + y_2^2 & y_2 & 1 \\ x_3^2 + y_3^2 & y_3 & 1 \end{vmatrix},\quad
O_y = \frac{1}{4\Delta} \begin{vmatrix} x_1 & x_1^2 + y_1^2 & 1 \\ x_2 & x_2^2 + y_2^2 & 1 \\ x_3 & x_3^2 + y_3^2 & 1 \end{vmatrix}
\]
餘弦定理 $c^2=a^2+b^2-2ab\cos C$；正弦定理 $\frac{a}{\sin A}=2R$；內切圓半徑 $r=\frac{\Delta}{s}$；外接圓 $R=\frac{abc}{4\Delta}$。

\subsection*{C. 四邊形}
凸四邊形邊長 $a,b,c,d$，$s=\frac{a+b+c+d}{2}$，已知 $\angle A+\angle C$：
面積 $=\sqrt{(s-a)(s-b)(s-c)(s-d) - abcd\cos^2\frac{A+C}{2}}$。圓內接四邊形時第二項為 0。
只知道一個角：用餘弦定理求對角線，再用海龍公式算兩個三角形。

\section{DP}

\textbf{DP 四步驟}：
1. 定義狀態：寫清楚 \texttt{dp[i][j]} 代表什麼 (\textbf{題目多一個條件 = 多一維})。
2. 轉移：「最後一步」是什麼？由哪些較小的狀態來？
3. 初始值：\texttt{dp[0]}、不可能的狀態設 $\pm$INF。
4. 計算順序：確保用到的狀態都已算好 (通常由小到大)。答案在哪一格？

\textbf{常見狀態}：前 $i$ 個 (線性)、區間 $[l,r]$ (區間)、拓樸序 (DAG)。
